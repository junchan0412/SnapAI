def eligible_ids(rows, minimum=10):
    seen = set()
    result = []
    for row in rows:
        key = row.get("id")
        if key is None or key in seen:
            continue
        if row.get("active", False) and row.get("score", 0) >= minimum:
            seen.add(key)
            result.append(key)
    return result
